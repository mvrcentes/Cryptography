def inverse_mod(k, p):
    """Returns the inverse of k modulo p.

    This function returns the only integer x such that (x * k) % p == 1.

    k must be non-zero and p must be a prime.
    """
    if k == 0:
    raise ZeroDivisionError('division by zero')

    if k < 0:
    # k ** -1 = p - (-k) ** -1  (mod p)
    return p - inverse_mod(-k, p)

    # Extended Euclidean algorithm.
    s, old_s = 0, 1
    t, old_t = 1, 0
    r, old_r = p, k

    while r != 0:
    quotient = old_r // r
    old_r, r = r, old_r - quotient * r
    old_s, s = s, old_s - quotient * s
    old_t, t = t, old_t - quotient * t

    gcd, x, y = old_r, old_s, old_t

    assert gcd == 1
    assert (k * x) % p == 1

    return x % p

def point_add(point1, point2):
    """Returns the result of point1 + point2 according to the group law."""
    assert is_on_curve(point1)
    assert is_on_curve(point2)

    if point1 is None:
    # 0 + point2 = point2
    return point2
    if point2 is None:
    # point1 + 0 = point1
    return point1

    x1, y1 = point1
    x2, y2 = point2

    if x1 == x2 and y1 != y2:
    # point1 + (-point1) = 0
    return None

    if x1 == x2:
    # This is the case point1 == point2.
    m = (3 * x1 * x1 + curve['a']) * inverse_mod(2 * y1, curve['p'])
    else:
    # This is the case point1 != point2.
    m = (y1 - y2) * inverse_mod(x1 - x2, curve['p'])

    x3 = m * m - x1 - x2
    y3 = y1 + m * (x3 - x1)
    result = (x3 % curve['p'],
          -y3 % curve['p'])

    assert is_on_curve(result)

    return result


def scalar_mult(k, point):
    """Returns k * point computed using the double and point_add algorithm."""

    return None

    if k & 1:

        result = point_add(result, addend)

    addend = point_add(addend, addend)

def extended_gcd(a, b):
   """
   Extended Euclidean Algorithm to find the greatest common divisor
   and coefficients x, y such that ax + by = gcd(a, b).
   """
   if a == 0:
       return (b, 0, 1)
   else:
       g, x, y = extended_gcd(b % a, a)
       return (g, y - (b // a) * x, x)

   """
   Compute the modular multiplicative inverse of a modulo m.
   Raises an exception if the modular inverse does not exist.
   """
   g, x, y = extended_gcd(a, m)
   if g != 1:
       raise Exception('Modular inverse does not exist')
   else:
       return x % m

def affine_decrypt(ciphertext, a, b):
   """Decrypt a message encrypted with the Affine Cipher using
   the given key components a and b."""
   alphabet = string.ascii_uppercase
   m = len(alphabet)
   plaintext = ''
   # Compute the modular multiplicative inverse of a
   a_inv = modular_inverse(a, m)
   # Iterate through each character in the ciphertext
   for char in ciphertext:
       # Check if the character is in the alphabet
       if char in alphabet:
           # If it's an alphabet letter, decrypt it
           # Find the index of the character in the alphabet
           c = alphabet.index(char)
           # Apply the decryption formula: a_inv * (c - b) mod m

           plaintext += alphabet[p]
       else:
           # If the character is not in the alphabet, keep it unchanged
           plaintext += char
   # Return the decrypted plaintext
   return plaintext

def extended_gcd(a, b):
    """Algoritmo de Euclides Extendido para encontrar el MCD y los coeficientes."""
    if a == 0:
        return (b, 0, 1)
    else:
        g, x, y = extended_gcd(b % a, a)
        return (g, y - (b // a) * x, x)

def modular_inverse(a, m):
    """Calcula el inverso modular de 'a' módulo 'm'."""
    g, x, _ = extended_gcd(a, m)
    if g != 1:
        return None  # No tiene inverso modular
    return x % m

def decrypt_affine(texto, a, b):
    """Descifra un texto cifrado con el cifrado Afín."""
    resultado = ""
    a_inv = modular_inverse(a, len(ALFABETO))
    if a_inv is None:
        return None  # Saltar si no tiene inverso

    for char in texto:
        if char in ALFABETO:
            indice = (a_inv * (ALFABETO.index(char) - b)) % len(ALFABETO)
            resultado += ALFABETO[indice]
        else:
            resultado += char  # Mantener otros caracteres
    return resultado

def contar_palabras_validas(texto, diccionario):

# 📌 Índice de coincidencia para estimar la longitud de la clave
def index_of_coincidence(text):
    frequencies = Counter(text)
    N = len(text)
    ic = sum(f * (f - 1) for f in frequencies.values()) / (N * (N - 1))
    return ic * len(ALPHABET)  # Normalizamos con el alfabeto español

# 📌 Estimar la longitud de la clave usando el índice de coincidencia
def estimate_key_length(ciphertext, max_length=16):
    best_length = 1
    best_ic = 0
    for key_length in range(1, max_length + 1):
        slices = [''.join(ciphertext[i::key_length]) for i in range(key_length)]
        avg_ic = sum(index_of_coincidence(s) for s in slices) / key_length
        if avg_ic > best_ic:
            best_ic = avg_ic
            best_length = key_length
    return best_length

# 📌 Obtener frecuencia de letras en un texto
def get_frequencies(text):
    total = len(text)
    return {letter: text.count(letter) / total for letter in ALPHABET}

# 📌 Calcular similitud con el español usando producto interno
def similarity_score(text):
    freqs = get_frequencies(text)
    return sum(freqs.get(letter, 0) * DISTRIBUCION_TEORICA.get(letter, 0) for letter in ALPHABET)

# 📌 Decodificar un texto cifrado con una clave dada
def decrypt(ciphertext, key):
    plaintext = ''
    key_length = len(key)
    for i, char in enumerate(ciphertext):
        p = (ALPHABET.index(char) - ALPHABET.index(key[i % key_length])) % len(ALPHABET)
        plaintext += ALPHABET[p]
    return plaintext

# 📌 Evaluar una clave en paralelo
def evaluate_key(args):
    key, ciphertext, threshold = args
    decrypted_text = decrypt(ciphertext, key)
    
    # Evaluar el número de palabras en español encontradas
    words_found = sum(1 for word in decrypted_text.split() if word.lower() in words)

    # Evaluar similitud con el español
    score = similarity_score(decrypted_text)

    # 📌 Imprimir el intento de clave
    print(f"🔍 Intentando clave: {key} | Similitud: {score:.2f} | Palabras encontradas: {words_found}")

    # Si encuentra muchas palabras válidas, devolver inmediatamente
    if words_found >= threshold:
        return key, decrypted_text, score, True  # Señal de que encontramos una clave válida

    return key, decrypted_text, score, False  # Clave no 100% confirmada

# 📌 Paralelizar la búsqueda de la mejor clave
def find_best_key_parallel(ciphertext, key_length, threshold=200, num_workers=4):
    best_key = None
    best_text = ""
    best_score = -float("inf")

    pool = multiprocessing.Pool(num_workers)

    # Generar todas las combinaciones posibles de la clave
    key_candidates = itertools.product(ALPHABET, repeat=key_length)

    # Procesar en paralelo
    for key, decrypted_text, score, found in pool.imap_unordered(
        evaluate_key, (( ''.join(k), ciphertext, threshold) for k in key_candidates)
    ):
        if found:  # Si encontramos una clave que parece correcta, detener
            pool.terminate()
            return key, decrypted_text
        
        if score > best_score:
            best_score = score
            best_key = key
            best_text = decrypted_text
    
    pool.close()
    pool.join()

    return best_key, best_text

# 📌 Programa principal
def main():
    # Cargar el texto cifrado
    ciphertext = load_ciphertext()

    # Estimar la longitud de la clave
    key_length = estimate_key_length(ciphertext)
    print(f"🔑 Longitud estimada de la clave: {key_length}")

    # Encontrar la mejor clave posible usando paralelización
    key, plaintext = find_best_key_parallel(ciphertext, key_length)

    print(f"\n🔓 Clave encontrada: {key}\n")
    print(f"📜 Texto descifrado:\n{plaintext[:500]}...")  # Mostrar primeras 500 letras

def descifrar_cesar(texto, shift):
    """Descifra un texto con un desplazamiento dado en el Cifrado César."""
    resultado = ""
    for char in texto:
        if char in ALFABETO:
            indice = (ALFABETO.index(char) - shift) % len(ALFABETO)
            resultado += ALFABETO[indice]
        else:
            resultado += char  # Mantener otros caracteres
    return resultado

def contar_palabras_validas(texto, diccionario):
    """Intenta encontrar palabras en el texto sin espacios comparando con un diccionario."""
    palabras_encontradas = 0
    for i in range(len(texto)):
        for j in range(i + 2, len(texto) + 1):  # Busca palabras de mínimo 2 letras
            if texto[i:j].lower() in diccionario:
                palabras_encontradas += 1
    return palabras_encontradas

def fuerza_bruta_cesar(mensaje):
    """Prueba todos los desplazamientos (0-26) y muestra resultados con colores."""
    
    mejor_shift = None
    mejor_texto = ""
    mejor_palabras = 0

    for shift in range(len(ALFABETO)):  
        texto_descifrado = descifrar_cesar(mensaje, shift)
        palabras_validas = contar_palabras_validas(texto_descifrado, words)

        # Determinar color basado en cantidad de palabras acertadas
        color = Fore.GREEN if palabras_validas > 5 else Fore.RED

            mejor_shift = shift

    # Mostrar el mejor resultado encontrado
    if mejor_shift is not None:

        print(f"{Fore.GREEN}🔓 Mensaje Descifrado: {mejor_texto}\n")
